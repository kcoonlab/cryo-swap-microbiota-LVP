########### CLASSIFY with greengenes2


#based on code here: 
#https://github.com/biocore/greengenes2/blob/main/scripts/classifier.sh#L31-L41
#and greengenes2 announcement here 
#https://forum.qiime2.org/t/introducing-greengenes2-2022-10/25291/6


#Extract appropriate reads from classifier based on Kozich primers, finished in ~ 10minutes
qiime feature-classifier extract-reads \
    --i-sequences 2022.10.backbone.full-length.fna.qza\
    --p-f-primer GTGCCAGCMGCCGCGGTAA --p-r-primer GGACTACHVGGGTWTCTAAT \
    --p-read-orientation both \
    --o-reads 2022.10.backbone.myprimers.fna.qza 
 


#Train classifier ~20 minutes

qiime feature-classifier fit-classifier-naive-bayes \
    --i-reference-reads 2022.10.backbone.myprimers.fna.qza  \
    --i-reference-taxonomy 2022.10.backbone.tax.qza \
    --o-classifier 2022.10.backbone.myprimers.nb.qza

#ok, so this is a trained classifier. how to use? 





#classify, based on  https://github.com/qiime2/docs/blob/master/source/tutorials/feature-classifier.rst

# Classify seqs, 10 mins to classify 
qiime feature-classifier classify-sklearn \
     --i-classifier 2022.10.backbone.myprimers.nb.qza \
     --i-reads ../dada2_all_paired/merged-paired-rep-seqs.qza \
     --o-classification taxonomy.myprimers.gg2.qza

qiime metadata tabulate \
     --m-input-file taxonomy.myprimers.gg2.qza \
     --o-visualization taxonomy.qzv


#export the tax table 
qiime tools export \
 --input-path taxonomy.myprimers.gg2.qza \
 --output-path ../../phyloseq
#this is the current taxonomy.qza file
